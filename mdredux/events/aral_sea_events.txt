add_namespace = aral_sea

# Dry year (fired by the yearly hydrology roll)
country_event = {
	id = aral_sea.1
	title = aral_sea.1.t
	desc = aral_sea.1.d
	picture = GFX_report_event_generic_construction
	is_triggered_only = yes
	option = {
		name = aral_sea.1.a
		trigger = { check_variable = { treasury > 0.03 } has_political_power = 15 }
		ai_chance = { base = 3 }
		set_temp_variable = { treasury_change = -0.03 }
		modify_treasury_effect = yes
		add_political_power = -15
	}
	option = {
		name = aral_sea.1.b
		ai_chance = { base = 1 }
		custom_effect_tooltip = aral_sea.1.b_tt
		hidden_effect = {
			subtract_from_variable = { aral_sea_restoration = aral_dry_loss }
			update_aral_sea = yes
		}
	}
}

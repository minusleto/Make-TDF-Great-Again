# Initialize after country selection; no KAZ country-history edit is required.
# Startup initialization and the monthly update run only while Kazakhstan owns state 717 (Kyzylorda).
on_actions = {
	on_startup = {
		effect = {
			KAZ = {
				if = {
					limit = { owns_state = 717 }
					update_aral_sea = yes
				}
			}
		}
	}
	on_monthly = {
		effect = {
			if = {
				limit = { original_tag = KAZ owns_state = 717 }
				aral_monthly_tick = yes
			}
		}
	}
}

# New-game initialization has global scope, so explicitly enter KAZ.
# The monthly pulse is KAZ-specific; the scripted tick handles the state-ownership gate.
on_actions = {
	on_startup = {
		effect = {
			KAZ = {
				update_aral_sea = yes
			}
		}
	}
	on_monthly_KAZ = {
		effect = {
			aral_monthly_tick = yes
		}
	}
}
